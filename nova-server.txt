// Speech to text for live voice when the phone browser cannot do it itself (same provider key, e.g. Groq Whisper).
app.post('/api/transcribe', express.raw({ type: () => true, limit: '8mb' }), async (q, r) => {
  try {
    if (E.STT_OFF === '1') return r.status(404).end();
    if (!Buffer.isBuffer(q.body) || q.body.length < 800) return r.status(400).json({ error: 'No audio received' });
    const type = String(q.headers['content-type'] || 'audio/webm').split(';')[0];
    const ext = /mp4|m4a|aac/.test(type) ? 'm4a' : /ogg/.test(type) ? 'ogg' : /wav/.test(type) ? 'wav' : 'webm';
    const lang = String(q.query.lang || '').toLowerCase();
    const providers = [];
    if (E.LLM_KEY) providers.push({ base: BASE, key: E.LLM_KEY });
    if (E.GROQ_API_KEY && E.GROQ_API_KEY !== E.LLM_KEY) providers.push({ base: secureProviderBase('GROQ_BASE', E.GROQ_BASE, 'https://api.groq.com/openai/v1'), key: E.GROQ_API_KEY });
    if (!providers.length) return r.status(503).json({ error: 'No speech provider is configured' });
    let last = { status: 502, text: 'No speech provider succeeded' };
    for (const provider of providers) {
      for (const model of [E.STT_MODEL || 'whisper-large-v3-turbo', 'whisper-large-v3']) {
        const form = new FormData();
        form.append('file', new Blob([q.body], { type }), 'speech.' + ext);
        form.append('model', model); form.append('response_format', 'json'); form.append('temperature', '0');
        if (/^[a-z]{2}$/.test(lang)) form.append('language', lang);
        try {
          const u = await fetch(provider.base.replace(/\/$/, '') + '/audio/transcriptions', {
            method: 'POST', headers: { Authorization: 'Bearer ' + provider.key },
            body: form, signal: AbortSignal.timeout(30000)
          });
          const text = await readBoundedText(u, 1024 * 1024);
          if (u.ok) {
            let j = {}; try { j = JSON.parse(text); } catch (e) {}
            const transcript = String(j.text || '').trim();
            if (transcript) return r.json({ text: transcript, provider: provider.base.includes('groq') ? 'groq' : 'primary' });
            last = { status: 502, text: 'Speech provider returned an empty transcript' };
          } else {
            last = { status: u.status, text };
          }
        } catch (e) { last = { status: 504, text: 'Speech provider timed out or was unreachable' }; }
        if (last.status !== 400 && last.status !== 404) break;
      }
    }
    r.status([400, 401, 402, 403, 429, 502, 503, 504].includes(last.status) ? last.status : 502)
      .json({ error: 'Speech service unavailable: ' + String(last.text).slice(0, 160) });
  } catch (e) { r.status(502).json({ error: 'Speech service unreachable' }); }
});

// Keep requests small (free tiers limit tokens per minute): last N messages, long old answers shortened.
const MAXMSG = Number(E.LLM_MAX_HISTORY) || 10;
function slim(body) {
  const m = Array.isArray(body.messages) ? body.messages : [];
  const sys = m.filter(x => x.role === 'system').slice(0, 1);
  const rest = m.filter(x => x.role !== 'system').slice(-MAXMSG).map((x, i, a) =>
    (x.role === 'assistant' && typeof x.content === 'string' && x.content.length > 1500 && i < a.length - 1)
      ? { ...x, content: x.content.slice(0, 1500) + ' ...' } : x);
  while (rest.length && rest[0].role === 'tool') rest.shift();
  return { ...body, messages: capImg([...sys, ...rest].map(flat)) };
}
// Text-only models need plain string content; vision models accept at most 3 images per request.
const isImg = p => p && p.type === 'image_url';
const flat = x => (Array.isArray(x.content) && !x.content.some(isImg)) ? { ...x, content: x.content.map(p => (p && p.text) || '').join('\n') } : x;
function capImg(msgs) {
  let n = 0;
  const out = msgs.slice();
  for (let i = out.length - 1; i >= 0; i--) {
    if (!Array.isArray(out[i].content)) continue;
    out[i] = { ...out[i], content: out[i].content.map(p => isImg(p) ? (++n <= 3 ? p : { type: 'text', text: '[earlier image omitted]' }) : p) };
  }
  return out;
}
// Free tiers limit tokens per minute (Groq free: 8000). Keep every request under a token budget:
// drop the archive packs first, then the oldest messages, then shorten the middle of the system prompt.
const BUDGET = Number(E.LLM_MAX_INPUT_TOKENS) || 5000;
const VISION = E.LLM_VISION_MODEL || 'qwen/qwen3.8-27b';          // the only model that accepts photos
const VBUDGET = Number(E.LLM_VISION_BUDGET) || 6500;
const imgT = m => Array.isArray(m.content) ? m.content.filter(isImg).length * 2048 : 0;   // each image counts as 2048 tokens
const hasImg = b => (b.messages || []).some(m => imgT(m) > 0);
const txt = m => typeof m.content === 'string' ? m.content : (Array.isArray(m.content) ? m.content.map(p => (p && p.text) || '').join(' ') : '');
const est = s => { let a = 0, b = 0; for (let i = 0; i < s.length; i++) (s.charCodeAt(i) < 128 ? a++ : b++); return a / 3.5 + b / 1.2; };
function fit(body, budget) {
  const m = Array.isArray(body.messages) ? body.messages.slice() : [];
  let sys = m.find(x => x.role === 'system'), liveTail = '';
  if (sys && typeof sys.content === 'string') { const li = sys.content.indexOf('\n\n== LIVE DATA =='); if (li > 0) { liveTail = sys.content.slice(li); sys = { ...sys, content: sys.content.slice(0, li) }; } }
  const rest = m.filter(x => x !== sys);
  const restT = () => rest.reduce((n, x) => n + est(txt(x)) + imgT(x) + 4, 0);
  const total = () => (sys ? est(txt(sys)) : 0) + est(liveTail) + restT();
  if (sys && typeof sys.content === 'string' && total() > budget)
    sys = { ...sys, content: sys.content.replace(/\[\[PACKS\]\][\s\S]*?\[\[\/PACKS\]\]/, '') };
  while (rest.length > 1 && total() > budget) rest.shift();
  while (rest.length && rest[0].role === 'tool') rest.shift();
  if (sys && typeof sys.content === 'string' && total() > budget) {
    const sysT = est(sys.content), want = Math.max(800, budget - restT() - est(liveTail));
    const keep = Math.max(1500, Math.floor(sys.content.length * want / sysT)), head = Math.floor(keep * 0.7);
    sys = { ...sys, content: sys.content.slice(0, head) + '\n...\n' + sys.content.slice(-(keep - head)) };
  }
  if (sys && typeof sys.content === 'string') sys = { ...sys, content: sys.content.replace(/\[\[\/?PACKS\]\]/g, '') + liveTail };
  return { ...body, messages: sys ? [sys, ...rest] : rest };
}

// Image jobs run in the background so Render's HTTP request timeout cannot cancel a free queue wait.
const imageJobs = new Map();
const imageJobHits = new Map();
function pruneImageJobs() {
  const now = Date.now();
  for (const [id, job] of imageJobs) if (now - job.createdAt > 15 * 60 * 1000) imageJobs.delete(id);
  if (imageJobs.size > 80) {
    const oldest = [...imageJobs.values()].sort((a, b) => a.createdAt - b.createdAt);
    for (const job of oldest.slice(0, imageJobs.size - 80)) imageJobs.delete(job.id);
  }
}

  // Existing optional paid provider is disabled unless explicitly opted in by the service owner.
  if (!freeOnly && E.ALLOW_PAID_IMAGE === 'true' && E.IMAGE_KEY && E.IMAGE_MODEL) {
    try {
      const u = await fetch(secureProviderBase('IMAGE_BASE', E.IMAGE_BASE, 'https://api.openai.com/v1') + '/images/generations', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json', Authorization: 'Bearer ' + E.IMAGE_KEY },
        body: JSON.stringify({ model: E.IMAGE_MODEL, prompt, size: '1024x1024', response_format: 'b64_json' }),
        signal: AbortSignal.timeout(60000)
      });
      let j = {}; try { j = JSON.parse(await readBoundedText(u, 16 * 1024 * 1024)); } catch (e) { if (u.ok) throw e; }
      const b = j.data && j.data[0] && j.data[0].b64_json;
      if (u.ok && b) return { image: 'data:image/png;base64,' + b, provider: 'configured-image-provider' };
    } catch (e) { console.warn('Optional image provider failed:', String(e && e.message || e).slice(0, 120)); }
  }
  if (E.IMAGE_OFF === '1') throw new Error('Image generation is disabled');
  const w = Math.min(1024, Number(body.width) || 768), h = Math.min(1024, Number(body.height) || 768);
  const seed = Number(body.seed) || Math.floor(Math.random() * 1e9);

      const u = await fetch(host + encodeURIComponent(prompt) + '?' + qs + '&model=' + model, {
        headers: E.POLLI_KEY ? { Authorization: 'Bearer ' + E.POLLI_KEY } : {},
        signal: AbortSignal.timeout(12000)
      });

      if (u.ok && /^image\//.test(type)) {

        if (buf.length > 1000) return { image: 'data:' + type + ';base64,' + buf.toString('base64'), provider: 'pollinations' };
      }
    } catch (e) { /* continue to the community fallback */ }
  }

  // prompts still leave NOVA and are processed by third-party workers.
  if (E.NOVA_ALLOW_AI_HORDE !== '1' || !E.HORDE_API_KEY) throw new Error('No approved image provider is currently configured');
  const base = secureProviderBase('HORDE_BASE', E.HORDE_BASE, 'https://aihorde.net/api/v2');
  const headers = { 'Content-Type': 'application/json', apikey: E.HORDE_API_KEY, 'Client-Agent': 'NovaAI:1.0:https://novaai-4auo.onrender.com' };

  });

  for (let attempt = 0; attempt < 270; attempt++) {

      headers, signal: AbortSignal.timeout(8000)
    });

    if (state.done || Number(state.finished) >= 1) { complete = true; break; }
  }
  if (!complete) throw new Error('Free image queue is still busy; please retry later');

    headers, signal: AbortSignal.timeout(12000)
  });

  // worker-controlled URLs from the server: that would create an SSRF boundary.
  const rawImage = String(generated);
  let encoded = rawImage;

  const dataMatch = rawImage.match(/^data:(image\/(?:webp|png|jpeg|gif));base64,([A-Za-z0-9+/]+={0,2})$/i);
  if (dataMatch) {
    declaredType = dataMatch[1].toLowerCase();
    encoded = dataMatch[2];
  }
  const maxImageBytes = 8 * 1024 * 1024;
  if (encoded.length > Math.ceil(maxImageBytes * 4 / 3) + 8 || !/^(?:[A-Za-z0-9+/]{4})*(?:[A-Za-z0-9+/]{2}==|[A-Za-z0-9+/]{3}=)?$/.test(encoded)) {
    throw new Error('Free community image exceeded the supported size or format');
  }
  const imageBuffer = Buffer.from(encoded, 'base64');
  if (imageBuffer.length < 1000 || imageBuffer.length > maxImageBytes) {
    throw new Error('Free community image exceeded the supported size');
  }
  const isWebp = imageBuffer.length >= 12 && imageBuffer.toString('ascii', 0, 4) === 'RIFF' && imageBuffer.toString('ascii', 8, 12) === 'WEBP';
  const isPng = imageBuffer.length >= 8 && imageBuffer.subarray(0, 8).equals(Buffer.from([137, 80, 78, 71, 13, 10, 26, 10]));
  const isJpeg = imageBuffer.length >= 3 && imageBuffer[0] === 0xff && imageBuffer[1] === 0xd8 && imageBuffer[2] === 0xff;
  const isGif = imageBuffer.length >= 6 && ['GIF87a', 'GIF89a'].includes(imageBuffer.toString('ascii', 0, 6));
  const detectedType = isWebp ? 'image/webp' : isPng ? 'image/png' : isJpeg ? 'image/jpeg' : isGif ? 'image/gif' : '';
  if (!detectedType || (rawImage.startsWith('data:') && detectedType !== declaredType)) {
    throw new Error('Free community image format was not supported');
  }
  return { image: 'data:' + detectedType + ';base64,' + imageBuffer.toString('base64'), provider: 'community-free' };
}

  const a = rateWindow(imageJobHits, imageRateKey, now, 10 * 60 * 1000);

  pruneImageJobs();
  if (imageJobs.size >= 80) return r.status(503).json({ error: 'The image queue is busy. Please retry shortly.' });

  const job = { id, ip: q.ip, userId: q.novaUser?.id || null, status: 'queued', image: null, error: null, provider: null, createdAt: now, updatedAt: now };
  imageJobs.set(id, job);
  setImmediate(async () => {
    job.status = 'processing'; job.updatedAt = Date.now();
    try {
      const result = await generateFreeImage(prompt, body);
      job.image = result.image; job.provider = result.provider; job.status = 'done';

      job.error = String(e && e.message || e).slice(0, 180);
      job.status = 'error';
      console.warn('Nova image job failed:', job.error);
    } finally { job.updatedAt = Date.now(); }
  });

// Cache only verified fresh results. On provider failure, report unavailable rather than presenting stale data as current.
async function cached(key, ttl, fn) {
  const c = lcache.get(key), n = Date.now();
  if (c && c.v !== undefined && n - c.t < ttl) return c.v;
  if (inflight.has(key)) return inflight.get(key);
  if (c && c.e && n - c.e < (c.backoff || 180000)) throw new Error(c.msg);
  const work = (async () => {
    try { const v = await fn(); lcache.set(key, { v, t: Date.now() }); lprune(); return v; }
    catch (e) {
      const retry = Number(e && e.retryAfter) * 1000;
      const backoff = Math.min(30 * 60 * 1000, Math.max(180000, retry || 0));
      lcache.set(key, { t: Date.now(), e: Date.now(), backoff, msg: String(e && e.message).slice(0, 80) });
      throw e;
    }
  })();
  inflight.set(key, work);
  try { return await work; }
  finally { if (inflight.get(key) === work) inflight.delete(key); }
}
// HTTP with timeout, retry + backoff, rate-limit (429/5xx) handling
async function lhttp(url, o = {}) {
  const tries = Math.max(1, Number(o.tries) || 2); let err;
  for (let i = 0; i < tries; i++) {
    try {
      const initial = safeRedirectTarget(url);
      const originalHost = initial.hostname.toLowerCase();
      const signal = AbortSignal.timeout(o.timeout || 5000);
      let target = initial, r;
      for (let hop = 0; hop <= 3; hop++) {
        r = await fetch(target, {
          method: o.method || 'GET',
          headers: { 'User-Agent': L_UA, Accept: o.accept || '*/*', ...(o.headers || {}) },
          body: o.body, redirect: 'manual', signal
        });
        if (![301, 302, 303, 307, 308].includes(r.status)) break;
        if ((o.method || 'GET').toUpperCase() !== 'GET' || hop === 3) {
          throw new Error('provider redirect blocked');
        }
        const location = r.headers.get('location');
        if (!location) throw new Error('provider redirect missing location');
        try { await r.body?.cancel(); } catch { /* release redirect response body */ }
        target = safeRedirectTarget(new URL(location, target).href, originalHost);
      }
      if (r.status === 429) {
        const e = new Error('HTTP 429'); e.status = 429;
        const ra = r.headers.get('retry-after'); e.retryAfter = Number(ra) || (ra ? Math.max(0, Math.ceil((Date.parse(ra) - Date.now()) / 1000)) : 0);
        throw e;
      }
      if (r.status >= 500) { err = new Error('HTTP ' + r.status); err.status = r.status; if (i < tries - 1) { await sleep(500 * (i + 1)); continue; } throw err; }
      if (!r.ok) { const e = new Error('HTTP ' + r.status); e.status = r.status; throw e; }
      const txt = await readBoundedText(r, 3 * 1024 * 1024);
      return o.json ? JSON.parse(txt) : txt;
    } catch (e) {
      err = e;
      if (e && (e.name === 'SyntaxError' || e.status === 429 || (e.status >= 400 && e.status < 500) || /redirect blocked|unsafe provider URL|invalid provider URL|response too large/.test(String(e.message)))) break;
      if (i < tries - 1) await sleep(350 * (i + 1));
    }
  }
  throw err || new Error('request failed');
}

  // Paid search providers are attempted sequentially: once one succeeds, later providers
  // are not called. This prevents one user request from triggering multiple paid searches.
  if (E.TAVILY_KEY) primary.push({
    name: 'tavily',
    run: async () => {
      const items = E.TAVILY_SHARED_CACHE_ALLOWED === '1'
        ? await cached('tv:' + (o.news ? 'news:' : 'general:') + q, 3 * 6e4, () => tavily(q, o))
        : await tavily(q, o);
      return { items, via: 'tavily' };
    }
  });
  if (E.BRAVE_KEY) {
    primary.push({
      name: 'brave-llm-context',
      run: async () => ({ items: await braveContext(q, o), via: 'brave-llm-context' })
    });
    primary.push({
      name: 'brave-web',
      run: async () => ({ items: await brave(q), via: 'brave' })
    });
  }
  const primaryResult = await firstSuccessful(primary);
  errs.push(...primaryResult.failures.map(x => x.provider));
  if (primaryResult.value) return primaryResult.value;

  // GDELT is the only keyless general web fallback until all other source licences are cleared.
  const fallback = await firstSuccessful([
    { name: 'gdelt', run: async () => ({ items: await gdelt(q, { n: 8, timespan: '7d' }), via: 'gdelt' }) }
  ]);
  errs.push(...fallback.failures.map(x => x.provider));
  if (fallback.value) return fallback.value;
  throw new Error('all web search providers failed (' + errs.join(', ') + ')');
}

function parseLists(html) {
  const out = [], stack = [];
  for (const tk of html.split(/(<\/?(?:ul|li)(?:\s[^>]*)?>)/i)) {
    if (/^<li/i.test(tk)) { if (stack.length) stack[stack.length - 1].kids = true; stack.push({ text: '', kids: false }); }

function rank(items, q) {
  const tk = toks(q), nowT = Date.now();
  items.forEach(it => {
    const age = it.date ? (nowT - Date.parse(it.date)) / 36e5 : null, h = host(it.url); let s = 0;
    if (age != null && age >= 0) s += Math.max(0, 40 - Math.log2(1 + age) * 5);
    it.quality = (OFFICIAL_HOSTS.test(h) || it.kind === 'official') ? 'primary' : (TOP_NEWS.test(h) || TOP_NEWS.test(it.publisher || '')) ? 'major' : 'other';
    s += it.quality === 'primary' ? 22 : it.quality === 'major' ? 14 : 0;
    s += Math.min(18, (it.also || []).length * 6) + jac(tk, toks(it.title + ' ' + (it.snippet || ''))) * 40;
    if (it.kind === 'reference') s -= 4; it._s = s;
  });
  return items.sort((a, b) => b._s - a._s);
}
